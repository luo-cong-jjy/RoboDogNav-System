import socket
import struct
import threading

# 需要转发的组播组和端口
GROUPS = [
    ("224.10.10.202", 6692),
    ("224.10.10.202", 7782),
    ("224.10.10.201", 6691),
    ("224.10.10.201", 7781),
]

# 接收组播的接口 (eth0)
RECV_IFACE = "10.21.33.106"
# 转发出去的接口 (eth1)
SEND_IFACE = "10.21.31.106"


def join_multicast_group(group, port, iface_ip):
    """创建并返回一个加入了组播组的 UDP socket"""
    sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM, socket.IPPROTO_UDP)
    sock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    sock.bind((group, port))  # 绑定到组播端口

    # 加入组播组
    mreq = struct.pack("4s4s", socket.inet_aton(group), socket.inet_aton(iface_ip))
    sock.setsockopt(socket.IPPROTO_IP, socket.IP_ADD_MEMBERSHIP, mreq)

    return sock


def forwarder(group, port):
    """从 eth0 接收组播并转发到 eth1"""
    recv_sock = join_multicast_group(group, port, RECV_IFACE)

    send_sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM, socket.IPPROTO_UDP)
    # 指定发送接口为 eth1
    send_sock.setsockopt(socket.IPPROTO_IP, socket.IP_MULTICAST_IF, socket.inet_aton(SEND_IFACE))

    print(f"[Relay] Forwarding {group}:{port} from {RECV_IFACE} → {SEND_IFACE}")

    while True:
        data, addr = recv_sock.recvfrom(65535)
        send_sock.sendto(data, (group, port))


def main():
    threads = []
    for group, port in GROUPS:
        t = threading.Thread(target=forwarder, args=(group, port), daemon=True)
        t.start()
        threads.append(t)

    print("Multicast relay running...")
    for t in threads:
        t.join()


if __name__ == "__main__":
    main()

import socket
import struct
import threading

# 需要转发的组播组和端口
GROUPS = [
    ("224.10.10.202", 6692),
    ("224.10.10.202", 7782),
    ("224.10.10.201", 6691),
    ("224.10.10.201", 7781),
]

# 接收组播的接口 (eth0)
RECV_IFACE = "10.21.33.106"
# 转发出去的接口 (eth1)
SEND_IFACE = "10.21.31.106"


def join_multicast_group(group, port, iface_ip):
    """创建并返回一个加入了组播组的 UDP socket"""
    sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM, socket.IPPROTO_UDP)
    sock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    sock.bind((group, port))  # 绑定到组播端口

    # 加入组播组
    mreq = struct.pack("4s4s", socket.inet_aton(group), socket.inet_aton(iface_ip))
    sock.setsockopt(socket.IPPROTO_IP, socket.IP_ADD_MEMBERSHIP, mreq)

    return sock


def forwarder(group, port):
    """从 eth0 接收组播并转发到 eth1"""
    recv_sock = join_multicast_group(group, port, RECV_IFACE)

    send_sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM, socket.IPPROTO_UDP)
    # 指定发送接口为 eth1
    send_sock.setsockopt(socket.IPPROTO_IP, socket.IP_MULTICAST_IF, socket.inet_aton(SEND_IFACE))

    print(f"[Relay] Forwarding {group}:{port} from {RECV_IFACE} → {SEND_IFACE}")

    while True:
        data, addr = recv_sock.recvfrom(65535)
        send_sock.sendto(data, (group, port))


def main():
    threads = []
    for group, port in GROUPS:
        t = threading.Thread(target=forwarder, args=(group, port), daemon=True)
        t.start()
        threads.append(t)

    print("Multicast relay running...")
    for t in threads:
        t.join()


if __name__ == "__main__":
    main()
